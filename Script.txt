local BotCode = [=[
    if not game:IsLoaded() then game.Loaded:Wait() end
    task.wait(3)

    local Players = game:GetService("Players")
    local TextChatService = game:GetService("TextChatService")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local TeleportService = game:GetService("TeleportService")
    local HttpService = game:GetService("HttpService")
    local PathfindingService = game:GetService("PathfindingService")
    local UserInputService = game:GetService("UserInputService")
    local TweenService = game:GetService("TweenService")
    local Workspace = game:GetService("Workspace")

    local LocalPlayer = Players.LocalPlayer
    local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

    -- === 1. ИНТЕРФЕЙС-МЕНЮ (С черной шапкой) ===
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "DonationBotGui"
    screenGui.ResetOnSpawn = false
    screenGui.Parent = PlayerGui

    local mainFrame = Instance.new("Frame")
    mainFrame.Size = UDim2.new(0, 160, 0, 65)
    mainFrame.Position = UDim2.new(0, 20, 0.5, -30)
    mainFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    mainFrame.BorderSizePixel = 0
    mainFrame.Parent = screenGui

    local uiCorner = Instance.new("UICorner")
    uiCorner.CornerRadius = UDim.new(0, 8)
    uiCorner.Parent = mainFrame

    local dragBar = Instance.new("Frame")
    dragBar.Size = UDim2.new(1, 0, 0, 20)
    dragBar.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    dragBar.BorderSizePixel = 0
    dragBar.Parent = mainFrame
    
    local dragCorner = Instance.new("UICorner")
    dragCorner.CornerRadius = UDim.new(0, 8)
    dragCorner.Parent = dragBar

    local dragBlock = Instance.new("Frame")
    dragBlock.Size = UDim2.new(1, 0, 0, 10)
    dragBlock.Position = UDim2.new(0, 0, 1, -10)
    dragBlock.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    dragBlock.BorderSizePixel = 0
    dragBlock.Parent = dragBar

    local dragTitle = Instance.new("TextLabel")
    dragTitle.Size = UDim2.new(1, 0, 1, 0)
    dragTitle.BackgroundTransparency = 1
    dragTitle.Text = "Drag Here"
    dragTitle.TextColor3 = Color3.fromRGB(150, 150, 150)
    dragTitle.TextSize = 12
    dragTitle.Font = Enum.Font.SourceSansBold
    dragTitle.Parent = dragBar

    local toggleButton = Instance.new("TextButton")
    toggleButton.Name = "ToggleButton"
    toggleButton.Size = UDim2.new(1, 0, 1, -20)
    toggleButton.Position = UDim2.new(0, 0, 0, 20)
    toggleButton.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
    toggleButton.Text = "Bot: ON"
    toggleButton.TextColor3 = Color3.fromRGB(255, 255, 255)
    toggleButton.TextSize = 18
    toggleButton.Font = Enum.Font.SourceSansBold
    toggleButton.Parent = mainFrame

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 8)
    btnCorner.Parent = toggleButton

    -- Перетаскивание
    local dragging, dragInput, dragStart, startPos
    dragBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = mainFrame.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    dragBar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    -- === 2. ПЕРЕМЕННЫЕ ===
    local isRunning = true
    local currentTarget = nil
    local agreedFlag = false
    local actuallyDonatedFlag = false
    local rejectedFlag = false
    local visitedPlayers = {}
    local hasClaimedBooth = false
    local myBoothPosition = nil 
    local BUBBLE_CHAT_DURATION = 5.2 

    local begPhrases = {
        "Pls robux donate me!",
        "Can you please donate a few robux?",
        "Please help me out with some robux!",
        "Pls donate robux bro, I really need it!"
    }
    local negativeWords = {"no", "nah", "sorry", "cant", "can't", "don't", "dont", "stop", "go away", "never", "0", "nop", "broke", "none"}
    local positiveWords = {"ok", "okay", "sure", "yes", "yeah", "lead", "where", "come", "alright", "ofc", "yep", "follow"}
    local donatedWords = {"donated", "done", "bought", "paid", "here", "thx"}

    -- === 2.5. АНТИ-СИДЕНИЕ НА СКАМЕЙКАХ ===
    task.spawn(function()
        while task.wait(0.2) do
            if not isRunning then continue end
            local char = LocalPlayer.Character
            if char then
                local hum = char:FindFirstChildOfClass("Humanoid")
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hum and hrp and hum.Sit then
                    hum.Sit = false
                    hum.Jump = true
                    -- Отпрыгиваем подальше от скамейки
                    hum:MoveTo(hrp.Position + hrp.CFrame.LookVector * 10)
                end
            end
        end
    end)

    -- === 3. РЕАЛИСТИЧНЫЙ ЧАТ И ШИФТЛОК ===
    local function typeAndSendMessage(message)
        local typeDelay = (string.len(message) / 12) + (math.random(10, 30) / 100)
        task.wait(typeDelay)
        pcall(function()
            if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
                local generalChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
                if generalChannel then generalChannel:SendAsync(message) end
            else
                local chatEvent = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
                if chatEvent then
                    local sayMessage = chatEvent:FindFirstChild("SayMessageRequest")
                    if sayMessage then sayMessage:FireServer(message, "All") end
                end
            end
        end)
    end

    local function smoothLookAt(targetPos)
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("Humanoid") then
            local hrp = char.HumanoidRootPart
            local hum = char.Humanoid
            if hum.MoveDirection.Magnitude < 0.1 then -- Только когда стоим
                local lookPos = Vector3.new(targetPos.X, hrp.Position.Y, targetPos.Z)
                local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
                local tween = TweenService:Create(hrp, tweenInfo, {CFrame = CFrame.lookAt(hrp.Position, lookPos)})
                tween:Play()
            end
        end
    end

    -- === 4. УМНЫЙ ОБХОД ПРЕПЯТСТВИЙ (PATHFINDING 3.0) ===
    local function smartMoveTo(targetPosition)
        local character = LocalPlayer.Character
        if not character then return end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not hrp then return end

        local path = PathfindingService:CreatePath({ AgentRadius = 3, AgentHeight = 5, AgentCanJump = true })
        local success = pcall(function() path:ComputeAsync(hrp.Position, targetPosition) end)

        if success and path.Status == Enum.PathStatus.Success then
            local waypoints = path:GetWaypoints()
            for i, waypoint in ipairs(waypoints) do
                if not isRunning then break end
                if i == 1 then continue end 
                
                if waypoint.Action == Enum.PathWaypointAction.Jump then humanoid.Jump = true end
                humanoid:MoveTo(waypoint.Position)
                
                local timeout = 0
                local startPos = hrp.Position
                while isRunning and timeout < 20 do
                    task.wait(0.1)
                    timeout = timeout + 1
                    
                    local dist2D = (Vector3.new(hrp.Position.X, 0, hrp.Position.Z) - Vector3.new(waypoint.Position.X, 0, waypoint.Position.Z)).Magnitude
                    if dist2D < 4 then break end 
                    
                    -- Если за 0.5 сек почти не сдвинулись - застряли, прыгаем!
                    if timeout % 5 == 0 then
                        if (hrp.Position - startPos).Magnitude < 1.5 then
                            humanoid.Jump = true
                        end
                        startPos = hrp.Position
                    end
                end
            end
        else
            humanoid:MoveTo(targetPosition)
        end
    end

    -- === 5. ТЕЛЕПОРТ К СТЕНДУ + ФОРС ПРОГРУЗКА ===
    local function claimBoothTeleport(attemptCount)
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("HumanoidRootPart") then return false end
        local hrp = character.HumanoidRootPart

        -- Если за 2 попытки не нашли, телепортируемся в центр карты, чтобы прогрузить мир (StreamingEnabled)
        if attemptCount == 2 or attemptCount == 5 then
            hrp.CFrame = CFrame.new(0, 10, 0)
            task.wait(1.5)
        end

        local targetPrompt = nil
        local targetPart = nil
        local shortestDist = math.huge

        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("ProximityPrompt") then
                local aText = string.lower(obj.ActionText or "")
                local oText = string.lower(obj.ObjectText or "")
                
                if string.find(aText, "claim") or string.find(oText, "unclaimed") or string.find(aText, "rent") then
                    if not string.find(aText, "donate") and not string.find(oText, "donate") then
                        -- Проверяем, что это не игрок
                        local model = obj:FindFirstAncestorOfClass("Model")
                        if model and model:FindFirstChild("Humanoid") then continue end

                        local part = obj:FindFirstAncestorWhichIsA("BasePart")
                        if not part and model then 
                            part = model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart", true) 
                        end

                        if part then
                            local dist = (hrp.Position - part.Position).Magnitude
                            if dist < shortestDist then
                                shortestDist = dist
                                targetPrompt = obj
                                targetPart = part
                            end
                        end
                    end
                end
            end
        end

        if targetPrompt and targetPart then
            -- МГНОВЕННЫЙ ТЕЛЕПОРТ К СТЕНДУ (КАК ТЫ И ПРОСИЛ)
            hrp.CFrame = targetPart.CFrame * CFrame.new(0, 2, 3.5)
            task.wait(0.5)
            
            -- Принудительно включаем кнопку, даже если она отключена
            targetPrompt.Enabled = true
            targetPrompt.MaxActivationDistance = 50

            if fireproximityprompt then
                fireproximityprompt(targetPrompt)
            else
                targetPrompt:InputHoldBegin()
                task.wait(0.3)
                targetPrompt:InputHoldEnd()
            end
            
            myBoothPosition = targetPart.Position
            task.wait(1.5)
            return true
        end

        return false
    end

    -- === 6. СМЕНА СЕРВЕРА ===
    local function serverHop()
        toggleButton.Text = "Hopping..."
        toggleButton.BackgroundColor3 = Color3.fromRGB(255, 100, 0)
        
        local q = (syn and syn.queue_on_teleport) or queue_on_teleport or (fluxus and fluxus.queue_on_teleport) or teleportqueue
        if q then pcall(function() q('pcall(function() loadstring(readfile("AutoForestBot.txt"))() end)') end) end
        
        task.wait(1)
        pcall(function()
            local req = request or http_request or (http and http.request) or syn.request
            if req then
                local res = req({Url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100", Method = "GET"})
                local data = HttpService:JSONDecode(res.Body)
                if data and data.data then
                    for _, s in ipairs(data.data) do
                        if s.playing < s.maxPlayers and s.id ~= game.JobId then
                            TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                            return
                        end
                    end
                end
            end
            TeleportService:Teleport(game.PlaceId, LocalPlayer)
        end)
    end

    -- === 7. ОБРАБОТКА ЧАТА ===
    local function containsWord(message, wordList)
        for _, word in ipairs(wordList) do
            if string.match(message, "%f[%w]" .. word .. "%f[%W]") then return true end
        end
        return false
    end

    local function onPlayerChatted(player, msg)
        if not isRunning or player ~= currentTarget then return end
        local lowerMsg = string.lower(msg)
        
        if containsWord(lowerMsg, negativeWords) then rejectedFlag = true; return end
        if containsWord(lowerMsg, positiveWords) then agreedFlag = true; return end
        if containsWord(lowerMsg, donatedWords) then actuallyDonatedFlag = true; return end
    end

    for _, player in ipairs(Players:GetPlayers()) do player.Chatted:Connect(function(msg) onPlayerChatted(player, msg) end) end
    Players.PlayerAdded:Connect(function(player) player.Chatted:Connect(function(msg) onPlayerChatted(player, msg) end) end)

    -- === 8. ГЛАВНАЯ ЛОГИКА ===
    local function startBeggingLoop()
        task.spawn(function()
            local claimAttempts = 0
            
            -- ЗАХВАТ СТЕНДА
            while not hasClaimedBooth and isRunning do
                toggleButton.Text = "Searching Booth..."
                toggleButton.BackgroundColor3 = Color3.fromRGB(0, 150, 200)
                
                claimAttempts = claimAttempts + 1
                if claimBoothTeleport(claimAttempts) then
                    hasClaimedBooth = true
                    toggleButton.Text = "Bot: ON"
                    toggleButton.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
                else
                    if claimAttempts >= 8 then -- Если 8 раз не смог найти, меняем сервер
                        toggleButton.Text = "Full! Hopping..."
                        toggleButton.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
                        task.wait(2)
                        serverHop()
                        return
                    end
                    task.wait(1.5)
                end
            end
            
            -- ОБХОД ИГРОКОВ
            while isRunning do
                local character = LocalPlayer.Character
                if not character or not character:FindFirstChild("HumanoidRootPart") then task.wait(1); continue end
                
                local closestPlayer = nil
                local shortestDistance = math.huge
                
                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer and not visitedPlayers[player.UserId] and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                        -- Игнорируем сидящих
                        local targetHum = player.Character:FindFirstChildOfClass("Humanoid")
                        if targetHum and targetHum.Sit then continue end

                        local dist = (character.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                        if dist < shortestDistance then
                            shortestDistance = dist
                            closestPlayer = player
                        end
                    end
                end
                
                -- Если все игроки кончились
                if not closestPlayer then
                    typeAndSendMessage("No more people here, hopping server...")
                    task.wait(2)
                    serverHop()
                    break
                end
                
                visitedPlayers[closestPlayer.UserId] = true
                currentTarget = closestPlayer
                agreedFlag = false
                rejectedFlag = false
                actuallyDonatedFlag = false
                
                -- Идем к игроку
                if closestPlayer.Character and closestPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    smartMoveTo(closestPlayer.Character.HumanoidRootPart.Position + Vector3.new(3, 0, 3))
                end
                
                -- ПРЕСЛЕДОВАНИЕ И ОБЩЕНИЕ
                for attempt = 1, 3 do
                    if not isRunning or currentTarget ~= closestPlayer or rejectedFlag or agreedFlag then break end
                    
                    if closestPlayer.Character and closestPlayer.Character:FindFirstChild("HumanoidRootPart") then
                        smoothLookAt(closestPlayer.Character.HumanoidRootPart.Position)
                    end
                    
                    typeAndSendMessage(begPhrases[math.random(1, #begPhrases)])
                    
                    local timer = 0
                    while timer < BUBBLE_CHAT_DURATION and isRunning and not rejectedFlag and not agreedFlag do
                        task.wait(0.1)
                        timer = timer + 0.1
                        
                        -- Бежим за игроком, если он уходит
                        if closestPlayer.Character and closestPlayer.Character:FindFirstChild("HumanoidRootPart") then
                            local targetPos = closestPlayer.Character.HumanoidRootPart.Position
                            local currentDist = (character.HumanoidRootPart.Position - targetPos).Magnitude
                            
                            if currentDist > 5.5 and currentDist < 60 then
                                character.Humanoid:MoveTo(targetPos)
                                if character.HumanoidRootPart.Velocity.Magnitude < 0.5 then
                                    character.Humanoid.Jump = true
                                end
                            elseif currentDist <= 5.5 then
                                character.Humanoid:MoveTo(character.HumanoidRootPart.Position)
                                smoothLookAt(targetPos)
                            end
                        end
                    end
                end

                -- ОТВОД К СТЕНДУ
                if agreedFlag and isRunning then
                    typeAndSendMessage("Follow me to my booth please!")
                    task.wait(1.5)
                    
                    if myBoothPosition then
                        smartMoveTo(myBoothPosition + Vector3.new(0, 0, 4.5))
                        smoothLookAt(myBoothPosition)
                    end
                    
                    local waitTimer = 0
                    while waitTimer < 15 and isRunning and not actuallyDonatedFlag do
                        task.wait(1)
                        waitTimer = waitTimer + 1
                        
                        if waitTimer % 4 == 0 and character and myBoothPosition then
                            local restlessMove = myBoothPosition + Vector3.new(math.random(-3, 3), 0, math.random(3, 5))
                            character.Humanoid:MoveTo(restlessMove)
                            if math.random(1, 3) == 1 then character.Humanoid.Jump = true end
                        end
                        
                        if waitTimer == 7 and currentTarget and currentTarget.Character then
                            local distToMe = (character.HumanoidRootPart.Position - currentTarget.Character.HumanoidRootPart.Position).Magnitude
                            if distToMe > 25 then 
                                smoothLookAt(currentTarget.Character.HumanoidRootPart.Position)
                                typeAndSendMessage("Are you coming?") 
                                task.wait(1.5)
                                smoothLookAt(myBoothPosition)
                            end
                        end
                    end

                    if actuallyDonatedFlag then
                        typeAndSendMessage("Thank you so much!!")
                        task.wait(2)
                    else
                        typeAndSendMessage("Well, maybe next time...")
                        task.wait(2)
                    end
                end
                
                task.wait(0.5)
            end
        end)
    end

    toggleButton.MouseButton1Click:Connect(function()
        isRunning = not isRunning
        if isRunning then
            toggleButton.Text = "Bot: ON"
            toggleButton.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
            startBeggingLoop()
        else
            toggleButton.Text = "Bot: OFF"
            toggleButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
            currentTarget = nil
        end
    end)

    startBeggingLoop()
]=]

pcall(function() writefile("AutoForestBot.txt", BotCode) end)
local q = (syn and syn.queue_on_teleport) or queue_on_teleport or (fluxus and fluxus.queue_on_teleport) or teleportqueue
if q then pcall(function() q('pcall(function() loadstring(readfile("AutoForestBot.txt"))() end)') end) end
loadstring(BotCode)()
