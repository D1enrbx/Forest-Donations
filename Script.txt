local function sendToServer(dataType, message, color)
    local data = {
        type = dataType,
        bot_name = LocalPlayer.Name,
        job_id = game.JobId,
        status = message or "",
        raised = currentRaised,
        uptime = math.floor((os.time() - botStartTime) / 60),
        message = message or "",
        color = color or "text-white",
        visited = stats_visited,
        agreed = stats_agreed,
        rejected = stats_rejected,
        ignored = stats_ignored
    }
    local req = (syn and syn.request) or http_request or request
    if req then
        pcall(function()
            req({Url = SERVER_URL, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = HttpService:JSONEncode(data)})
        end)
    end
end

local isRunning = true

local actuallyDonatedFlag = false
local rejectedFlag = false

local begPhrases = {
    "Pls robux donate me!", "Can anyone please donate a few robux? Saving up for an item!",
    "Please help me out with some robux if you can :(", "Pls donate robux bro, I really need it!",
    "Any small donation helps a lot! Thank you <3"
}
local negativeWords = {"no", "nah", "sorry", "cant", "can't", "don't", "dont", "stop", "go away", "never", "0", "nop", "broke", "none", "busy", "nope"}
local positiveWords = {"ok", "okay", "sure", "yes", "yeah", "lead", "where", "come", "alright", "ofc", "yep", "follow", "show", "lets"}

local function typeAndSendMessage(message)
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            TextChatService.TextChannels:FindFirstChild("RBXGeneral"):SendAsync(message)
        else
            ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents"):FindFirstChild("SayMessageRequest"):FireServer(message, "All")
        end
    end)
end

-- === ДВИЖЕНИЕ ===
local function aggressiveMoveTo(targetPos)
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    local timeout = 0
    local lastPos = hrp.Position
    while isRunning do
        if (hrp.Position - targetPos).Magnitude <= 5 then break end
        pcall(function() hum:MoveTo(targetPos) end)
        timeout = timeout + 1
        if timeout % 10 == 0 then
            if (hrp.Position - lastPos).Magnitude < 1 then pcall(function() hum.Jump = true end) end
            lastPos = hrp.Position
        end
        if timeout > 100 then break end
        task.wait(0.1)
    end
end

-- === ЗАХВАТ СТЕНДА ===
local function claimBoothUltimate()
    setStatus("Searching for booth...")
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if not hrp then return false end

    local targetPrompt, targetPart = nil, nil
    pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("ProximityPrompt") then
                local t1, t2, t3 = string.lower(obj.ActionText or ""), string.lower(obj.ObjectText or ""), string.lower(obj.Name or "")
                if string.find(t1, "claim") or string.find(t2, "claim") or string.find(t3, "claim") then
                    targetPrompt = obj
                    targetPart = obj.Parent
                    if targetPart and targetPart:IsA("Attachment") then targetPart = targetPart.Parent end
                    break 
                end
            end
        end
    end)

    if targetPrompt and targetPart and targetPart:IsA("BasePart") then
        setStatus("Teleporting to booth...")
        pcall(function()
            hrp.CFrame = targetPart.CFrame * CFrame.new(0, 3, 3)
            task.wait(0.3)
            hrp.CFrame = targetPart.CFrame * CFrame.new(0, 3, 3)
        end)
        setStatus("Activating booth...")
        task.wait(0.5)
        pcall(function()
            targetPrompt.Enabled = true
            targetPrompt.MaxActivationDistance = 999
            if fireproximityprompt then fireproximityprompt(targetPrompt) else
                targetPrompt:InputHoldBegin() task.wait(0.5) targetPrompt:InputHoldEnd()
            end
        end)
        myBoothPosition = targetPart.Position
        task.wait(1.5)
        return true
    end
    return false
end

local function serverHop()
    setStatus("Hopping server...")
    sendToServer("notification", "Hopping to a new server...", "text-orange-400")
    local q = (syn and syn.queue_on_teleport) or queue_on_teleport or (fluxus and fluxus.queue_on_teleport) or teleportqueue
    if q then pcall(function() q('loadstring(game:HttpGet("https://raw.githubusercontent.com/D1enrbx/Forest-Donations/refs/heads/main/Script.txt"))()') end) end
    task.wait(1)
    pcall(function()
        local req = request or http_request or (http and http.request) or syn.request
        local res = req({Url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100", Method = "GET"})
        local data = HttpService:JSONDecode(res.Body)
        for _, s in ipairs(data.data) do
            if s.playing < s.maxPlayers and s.id ~= game.JobId then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                return
            end
        end
        TeleportService:Teleport(game.PlaceId, LocalPlayer)
    end)
end

-- === ГЛАВНЫЙ ЦИКЛ ===
local function startBeggingLoop()
    task.spawn(function()
        while not hasClaimedBooth and isRunning do
            if claimBoothUltimate() then
                hasClaimedBooth = true
                setStatus("Booth Claimed!")
                sendToServer("notification", "Successfully claimed stand!", "text-green-400")
            else
                task.wait(2)
                serverHop()
                return
            end
        end
        
        while isRunning do
            pcall(function()
                local char = LocalPlayer.Character
                if not char or not char:FindFirstChild("HumanoidRootPart") then return end
                
                local closestPlayer = nil
                local shortestDist = math.huge
                
                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer and not visitedPlayers[player.UserId] and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                        local dist = (char.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestPlayer = player
                        end
                    end
                end
                
                if not closestPlayer then
                    setStatus("All visited, hopping...")
                    typeAndSendMessage("Hopping server...")
                    task.wait(2)
                    serverHop()
                    isRunning = false
                    return
                end
                
                visitedPlayers[closestPlayer.UserId] = true
                currentTarget = closestPlayer
                agreedFlag = false
                rejectedFlag = false
                actuallyDonatedFlag = false
                
                -- ОБНОВЛЯЕМ СТАТИСТИКУ VISITED
                stats_visited = stats_visited + 1
                
                setStatus("Walking to player")
                aggressiveMoveTo(closestPlayer.Character.HumanoidRootPart.Position + Vector3.new(3, 0, 3))
                
                for attempt = 1, 3 do
                    if not isRunning or rejectedFlag or agreedFlag then break end
                    pcall(function() char.HumanoidRootPart.CFrame = CFrame.lookAt(char.HumanoidRootPart.Position, Vector3.new(closestPlayer.Character.HumanoidRootPart.Position.X, char.HumanoidRootPart.Position.Y, closestPlayer.Character.HumanoidRootPart.Position.Z)) end)
                    
                    setStatus("Asking (" .. attempt .. "/3)")
                    typeAndSendMessage(begPhrases[math.random(1, #begPhrases)])
                    
                    local timer = 0
                    while timer < 5 and isRunning and not rejectedFlag and not agreedFlag do
                        task.wait(0.1)
                        timer = timer + 0.1
                        if closestPlayer.Character and closestPlayer.Character:FindFirstChild("HumanoidRootPart") then
                            local tPos = closestPlayer.Character.HumanoidRootPart.Position
                            if (char.HumanoidRootPart.Position - tPos).Magnitude > 6 then
                                pcall(function() char.Humanoid:MoveTo(tPos) end)
                            end
                        end
                    end
                end

                if agreedFlag and isRunning then
                    -- ОБНОВЛЯЕМ СТАТИСТИКУ AGREED
                    stats_agreed = stats_agreed + 1
                    sendToServer("notification", "Player agreed to follow!", "text-green-300")
                    
                    setStatus("Leading to booth!")
                    typeAndSendMessage("Follow me to my booth please!")
                    task.wait(1.5)
                    
                    if myBoothPosition then aggressiveMoveTo(myBoothPosition + Vector3.new(0, 0, 4.5)) end
                    
                    local waitTimer = 0
                    while waitTimer < 20 and isRunning and not actuallyDonatedFlag do
                        task.wait(1)
                        waitTimer = waitTimer + 1
                        setStatus("Waiting for donate (" .. waitTimer .. "s)")
                    end

                    if actuallyDonatedFlag then
                        setStatus("Donation received!")
                        typeAndSendMessage("Thank you so much!! <3")
                        task.wait(2)
                    else
                        setStatus("Moving on...")
                        typeAndSendMessage("Maybe next time...")
                        task.wait(1)
                    end
                elseif rejectedFlag then
                    -- ОБНОВЛЯЕМ СТАТИСТИКУ REJECTED
                    stats_rejected = stats_rejected + 1
                    sendToServer("notification", "Player rejected.", "text-red-400")
                else
                    -- ОБНОВЛЯЕМ СТАТИСТИКУ IGNORED
                    stats_ignored = stats_ignored + 1
                    sendToServer("notification", "Player ignored.", "text-gray-400")
                end
            end)
            task.wait(0.5)
        end
    end)
end

for _, player in ipairs(Players:GetPlayers()) do 
    player.Chatted:Connect(function(msg)
        local lMsg = string.lower(msg)
        if containsWord(lMsg, negativeWords) then rejectedFlag = true end
        if containsWord(lMsg, positiveWords) then agreedFlag = true end
    end) 
end

    player.Chatted:Connect(function(msg)
        local lMsg = string.lower(msg)
        if containsWord(lMsg, negativeWords) then rejectedFlag = true end
        if containsWord(lMsg, positiveWords) then agreedFlag = true end
    end) 
end)
