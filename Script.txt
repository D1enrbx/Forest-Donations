local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local userId = tostring(LocalPlayer.UserId)
local playerName = LocalPlayer.Name

local SERVER_URL = "https://d1enrbx.pythonanywhere.com/verify"
local REPORT_URL = "https://d1enrbx.pythonanywhere.com/report"
local SAVE_FILE = "ForestDonations_Key.txt"

-- Функция проверки ключа
local function verifyKey(key)
    local url = SERVER_URL .. "?key=" .. HttpService:UrlEncode(key) .. "&uid=" .. HttpService:UrlEncode(userId)
    local success, response = pcall(function()
        if syn and syn.request then return syn.request({Url = url, Method = "GET"}).Body
        elseif http_request then return http_request({Url = url, Method = "GET"}).Body
        elseif request then return request({Url = url, Method = "GET"}).Body
        else return game:HttpGet(url) end
    end)
    if success then
        local ok, data = pcall(function() return HttpService:JSONDecode(response) end)
        if ok then return data end
    end
    return {status = "error", message = "Ошибка соединения с сервером"}
end

-- Отправка статистики
local function sendTelemetry(key)
    local earned = 0
    if _G.ForestEarned and type(_G.ForestEarned) == "number" then
        earned = _G.ForestEarned
    else
        local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
        if leaderstats then
            for _, stat in ipairs(leaderstats:GetChildren()) do
                local n = string.lower(stat.Name)
                if (n:find("raised") or n:find("robux") or n:find("donated") or n:find("earned")) and (stat:IsA("NumberValue") or stat:IsA("IntValue")) then
                    earned = stat.Value
                end
            end
        end
    end

    local booth = _G.ForestCurrentBooth or "Активен"

    local data = {
        key = key,
        uid = userId,
        name = playerName,
        earned = earned,
        booth = booth
    }
    local body = HttpService:JSONEncode(data)
    pcall(function()
        if syn and syn.request then
            syn.request({Url = REPORT_URL, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = body})
        elseif http_request then
            http_request({Url = REPORT_URL, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = body})
        elseif request then
            request({Url = REPORT_URL, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = body})
        end
    end)
end

-- Запуск основного скрипта
local function startMainScript(key)
    ScreenGui:Destroy()

    -- Отправка телеметрии каждые 10 секунд
    task.spawn(function()
        while true do
            sendTelemetry(key)
            task.wait(10)
        end
    end)

    print("[Forest Bot] Авторизация успешна. Загрузка основного скрипта...")

    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/D1enrbx/Forest-Donations/refs/heads/main/Script.txt"))()
    end)
    if not success then
        warn("[Forest Bot] Ошибка загрузки скрипта: " .. tostring(err))
    end
end

ActivateBtn.MouseButton1Click:Connect(function()
    local key = TextBox.Text
    if key == "" then
        StatusLabel.Text = "⚠️ Ошибка: Поле ключа пустое!"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 90, 90)
        return
    end

    StatusLabel.Text = "🔄 Проверка ключа на сервере..."
    StatusLabel.TextColor3 = Color3.fromRGB(0, 255, 102)

    local res = verifyKey(key)
    if res.status == "success" then
        StatusLabel.Text = "✅ Успешно! Запуск..."
        StatusLabel.TextColor3 = Color3.fromRGB(50, 255, 150)
        if writefile then writefile(SAVE_FILE, key) end
        task.wait(0.8)
        startMainScript(key)
    else
        StatusLabel.Text = "❌ " .. (res.message or "Неверный ключ или лимит исчерпан!")
        StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
    end
end)

if currentKey ~= "" then
    task.spawn(function()
        local res = verifyKey(currentKey)
        if res.status == "success" then
            startMainScript(currentKey)
        end
    end)
end
